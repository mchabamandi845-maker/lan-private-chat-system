// Instant LAN Chat using Web BroadcastChannel API
const channel = new BroadcastChannel('lan_private_chat');

// Listen for incoming LAN messages from other browser tabs / windows on network
channel.onmessage = (event) => {
    const { sender, payload, signature } = event.data;
    console.log(`[LAN RX] ${sender}:`, payload);
    renderMessage(sender, payload);
};

// Send message to all peers on the same origin / local web server

        payload: text,
        timestamp: Date.now()
    };
    channel.postMessage(packet);
}

Node.js Zero-Dependency Local Broadcast Server

server.js

Run this script on any computer on your Wi-Fi/Ethernet router to create a central LAN server hub:

// Save as lan_server.js -> Run: node lan_server.js
const { WebSocketServer } = require('ws');
const wss = new WebSocketServer({ port: 8080 });

console.log('⚡ LAN Chat Hub listening on ws://0.0.0.0:8080');

wss.on('connection', (ws, req) => {
    const clientIp = req.socket.remoteAddress;
    console.log(`[+] Peer connected from LAN IP: ${clientIp}`);

    ws.on('message', (message) => {
        // Broadcast packet to all connected LAN peers
        wss.clients.forEach(client => {
            if (client.readyState === 1) {
                client.send(message.toString());
            }
        });
    });
});

Python UDP Peer-to-Peer Multicast Discovery

peer.py

Pure peer-to-peer UDP broadcast script. No central server needed!

# Save as peer.py -> Run: python3 peer.py
import socket
import threading

PORT = 5005
BROADCAST_IP = '255.255.255.255'

# Receiver Thread
def listen():
    sock = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    sock.bind(('', PORT))
    while True:
        data, addr = sock.recvfrom(1024)
        print(f"\n[Packet from {addr[0]}]: {data.decode('utf-8')}\n> ", end="")

threading.Thread(target=listen, daemon=True).start()

# Sender Loop
sender = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
sender.setsockopt(socket.SOL_SOCKET, socket.SO_BROADCAST, 1)

while True:
    msg = input("> ")
    sender.sendto(msg.encode('utf-8'), (BROADCAST_IP, PORT))

Go Libp2p with mDNS Auto-Discovery

main.go

Enterprise-grade encrypted p2p networking with automatic LAN local peer discovery:

package main

import (
    "context"
    "fmt"
    "github.com/libp2p/go-libp2p"
    "github.com/libp2p/go-libp2p/p2p/discovery/mdns"
)

type discoveryNotifee struct{}
func (n *discoveryNotifee) HandlePeerFound(pi peer.AddrInfo) {
    fmt.Printf("[+] Discovered LAN Peer: %s\n", pi.ID.String())
}

func main() {
    host, _ := libp2p.New(libp2p.ListenAddrStrings("/ip4/0.0.0.0/tcp/0"))
    fmt.Printf("Node running on ID: %s\n", host.ID())

    // Setup mDNS Service for local LAN discovery
    ser := mdns.NewMdnsService(host, "cryptchat-lan", &discoveryNotifee{})
    ser.Start()

    select {} // Keep running
}
